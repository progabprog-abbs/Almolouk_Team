def resource_path(relative_path):
    if hasattr(sys, "_MEIPASS"):
        return os.path.join(sys._MEIPASS, relative_path)
    p1 = os.path.join(APP_ROOT, relative_path)
    if os.path.exists(p1):
        return p1
    p2 = os.path.join(APP_ROOT, "assets", relative_path)
    if os.path.exists(p2):
        return p2
    return p1

# مفتاح التشفير الداخلي المعتمد على مكتبات بايثون القياسية (متوافق 100% مع Android و Windows)
_SECRET_KEY = hashlib.sha256(b"Almolouk_Team_2026_Moola_SecureKey").digest()


class DatabaseManager:
    def __init__(self, db_name="almolouk_pro.db"):
        self.db_name = db_name
        self.init_db()

    def get_connection(self):
        return sqlite3.connect(self.db_name, check_same_thread=False)

    def encrypt_text(self, plain_text: str) -> str:
        """تشفير النصوص والرسائل والإشعارات بدون الحاجة لمكتبة cryptography"""
        if not plain_text:
            return ""
        raw = str(plain_text).encode("utf-8")
        xored = bytes(b ^ k for b, k in zip(raw, itertools.cycle(_SECRET_KEY)))
        return base64.urlsafe_b64encode(xored).decode("utf-8")

    def decrypt_text(self, cipher_text: str) -> str:
        """فك تشفير النصوص والرسائل المشفرة"""
        if not cipher_text:
            return ""
        try:
            xored = base64.urlsafe_b64decode(str(cipher_text).encode("utf-8"))
            raw = bytes(b ^ k for b, k in zip(xored, itertools.cycle(_SECRET_KEY)))
            return raw.decode("utf-8")
        except Exception:
            return str(cipher_text)

    def init_db(self):
        conn = self.get_connection()
        cursor = conn.cursor()

        # 1. جدول المستخدمين والصلاحيات
        cursor.execute("""

                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                password TEXT NOT NULL,
                role TEXT NOT NULL DEFAULT 'member'
            )
        """)

        # 2. جدول بيانات أعضاء فريق الملوك
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS members (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,

                position TEXT,
                jersey_number TEXT,
                age TEXT,
                notes TEXT,
                created_at TEXT
            )
        """)

        # 3. جدول الإشعارات الإدارية

            CREATE TABLE IF NOT EXISTS notifications (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                body TEXT NOT NULL,
                notif_type TEXT DEFAULT 'عام',
                sender TEXT,
                created_at TEXT

        """)

        # 4. جدول رسائل الدردشة والصوتيات
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS chat_messages (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                sender TEXT NOT NULL,

                msg_type TEXT DEFAULT 'text',

        # 5. جدول الإعدادات السحابية والترخيص
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS app_settings (
                setting_key TEXT PRIMARY KEY,
                setting_value TEXT
            )
        """)

        # إضافة حساب المدير الافتراضي إن لم يكن موجوداً
        cursor.execute("SELECT COUNT(*) FROM users WHERE username = 'admin'")
        if cursor.fetchone()[0] == 0:

    def verify_user(self, username: str, password: str):

        conn.close()

    def check_license_status(self, trial_days: int = 30):
        """التحقق من صلاحية الترخيص ومنع التلاعب بتاريخ الجهاز"""
        now = datetime.now()
        now_str = now.strftime("%Y-%m-%d %H:%M:%S")

        enc_expiry = self.get_setting("lic_expiry")
        enc_last_seen = self.get_setting("lic_last_seen")

        if not enc_expiry:
            expiry_dt = now + timedelta(days=trial_days)
            self.set_setting("lic_expiry", self.encrypt_text(expiry_dt.strftime("%Y-%m-%d %H:%M:%S")))
            self.set_setting("lic_last_seen", self.encrypt_text(now_str))
            return True, f"نسخة مفعلة ({trial_days} يوم متبقي)"

        try:
            expiry_dt = datetime.strptime(self.decrypt_text(enc_expiry), "%Y-%m-%d %H:%M:%S")
            if enc_last_seen:
                last_seen_dt = datetime.strptime(self.decrypt_text(enc_last_seen), "%Y-%m-%d %H:%M:%S")
                if now < last_seen_dt - timedelta(minutes=10):
                    return False, "تم اكتشاف تلاعب بساعة الجهاز! يرجى ضبط الوقت أو إدخال مفتاح التجديد."

            if now > expiry_dt:
                return False, "انتهت مدة صلاحية النسخة. يرجى التواصل مع المطور للتجديد."

            self.set_setting("lic_last_seen", self.encrypt_text(now_str))
            remaining = (expiry_dt - now).days
            return True, f"الترخيص ساري ({remaining} يوم متبقي)"
        except Exception:
            return False, "بيانات الترخيص غير صالحة."

            meta = header_part.replace("[DUR:", "")

            expiry_str = parts[1] if len(parts) > 1 else "PERMANENT"

            is_expired = False
            expiry_display = f"المدة: {duration_label}"
            if expiry_str != "PERMANENT":
                try:
                    exp_dt = datetime.strptime(expiry_str, "%Y-%m-%d %H:%M")

                    expiry_display = f"المدة: {duration_label} (ينتهي: {expiry_str})"

        text_align="left",

        try:
            db.cursor.execute("SELECT name, phone FROM members WHERE phone IS NOT NULL AND phone != '' AND phone != '123'")

            if not rows:
                return [ft.dropdown.Option(key="", text="لا يوجد أعضاء مسجلون بأرقام هواتف", disabled=True)]
            return [ft.dropdown.Option(key=str(row[1]), text=f"{row[0]} ({row[1]})") for row in rows]

            return [ft.dropdown.Option(key="", text="خطأ في قراءة البيانات", disabled=True)]
